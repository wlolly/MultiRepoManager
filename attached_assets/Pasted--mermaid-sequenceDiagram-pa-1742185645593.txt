KEY,username VARCHAR(255) UNIQUE NOT NULL,password_hash VARCHAR(255) NOT NULL,created_at TIMESTAMPTZ DEFAULT NOW());-- 登录验证表CREATE TABLE login_verifications (verification_id UUID PRIMARY KEY,user_id UUID REFERENCES users(user_id) NULL, -- 允许空值表示无效用户created_at TIMESTAMPTZ DEFAULT NOW(),expires_at TIMESTAMPTZ NOT NULL,is_used BOOLEAN DEFAULT FALSE);```---### **后端实现（Node.js + TypeScript）**#### 1. 登录请求处理```typescript// 登录接口app.post('/api/login', async (req, res) => {const { username, password } = req.body;// 验证用户是否存在const user = await db.users.findOne({ where: { username },select: ['user_id', 'password_hash']});// 无论是否存在都生成验证记录const verificationId = crypto.randomUUID();const expiresAt = new Date(Date.now() + 15 * 60 * 1000); // 15分钟过期await db.login_verifications.create({data: {verification_id: verificationId,user_id: user?.user_id || null, // 关键设计：有效用户关联ID，无效设为nullexpires_at: expiresAt}});// 统一返回验证ID（不暴露用户是否存在的信息）res.json({ verificationId });});```#### 2. 验证接口```typescriptapp.post('/api/verify', async (req, res) => {const { verificationId } = req.body;const verification = await db.login_verifications.findUnique({where: { verification_id: verificationId },include: { user: true }});// 验证规则检查const isValid = verification && !verification.is_used &&new Date() < verification.expires_at &&verification.user_id !== null; // 关键检查点if (isValid) {// 标记验证记录为已使用await db.login_verifications.update({where: { verification_id: verificationId },data: { is_used: true }});// 生成会话Tokenconst authToken = generateAuthToken(verification.user);res.json({ success: true, token: authToken });}else {res.status(401).json({ success: false,error: "Invalid verification ID" // 统一错误提示});}});```---### **前端实现（React + TypeScript）**#### 1. 登录流程```tsxconst LoginForm = () => {const [step, setStep] = useState<'init' | 'verify'>('init');const [verificationId, {// 错误处理}};return (<div>{step === 'init' && <Form onSubmit={handleSubmit} />}{step === 'verify' && <VerificationForm onVerify={handleVerify} />}</div>);};```---### **安全增强措施**1. **防暴力破解**```typescript// 在验证接口添加速率限制app.use('/api/verify', rateLimit({windowMs: 15 * 60 * 1000, // 15分钟max: 5, // 每个IP最多5次尝试handler: (req, res) => {res.status(429).json({ error: "Too many verification attempts" });}}));```2. **自动清理无效记录**```sql-- 每天清理过期记录CREATE OR REPLACE FUNCTION clean_expired_verifications()RETURNS VOID AS $$BEGINDELETE FROM login_verifications WHERE expires_at < NOW() - INTERVAL '1 day';END;$$ LANGUAGE plpgsql;-- 设置定时任务（每天凌晨执行）SELECT cron.schedule('0 0 * * *',$$SELECT clean_expired_verifications()$$);```---###
