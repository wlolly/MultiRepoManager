### **数据一致性验证规范**

#### **一、数据库关系验证**
```yaml
# 必须检查的数据库约束
database_checks:
  - 类型: 外键验证
    规则: |
      SELECT 
        tc.table_name, 
        kcu.column_name, 
        ccu.table_name AS foreign_table,
        ccu.column_name AS foreign_column 
      FROM 
        information_schema.table_constraints AS tc 
        JOIN information_schema.key_column_usage AS kcu
          ON tc.constraint_name = kcu.constraint_name
        JOIN information_schema.constraint_column_usage AS ccu
          ON ccu.constraint_name = tc.constraint_name
      WHERE 
        tc.constraint_type = 'FOREIGN KEY'
        
  - 类型: ORM映射验证
    示例代码: 
      ```python
      # Django示例：检查模型与数据库一致性
      python manage.py check --deploy --fail-level WARNING
      # SQLAlchemy示例
      from alembic import command
      from alembic.config import Config
      alembic_cfg = Config("alembic.ini")
      command.check(alembic_cfg)
      ```

#### **二、跨层命名一致性检查**
```typescript
// 自动验证三端命名规则的脚本示例
interface NamingRules {
  dbPattern: 'snake_case';
  apiPattern: 'camelCase';
  uiPattern: 'PascalCase';
}

const verifyNaming = (layer: 'db'|'api'|'ui', key: string) => {
  const patterns = {
    db: /^[a-z]+(_[a-z]+)*$/,
    api: /^[a-z]+([A-Z][a-z]*)*$/,
    ui: /^[A-Z][a-zA-Z]*$/
  };
  
  if (!patterns[layer].test(key)) {
    throw new Error(`命名规范违反：${layer}层键名 ${key}`);
  }
}

// 自动生成对应关系表
const generateMapping = (dbName: string) => {
  return {
    db: dbName,
    api: _.camelCase(dbName),
    ui: _.upperFirst(_.camelCase(dbName))
  };
}
```

SELECT key_name, COUNT(*) 
FROM translation_keys 
GROUP BY key_name 
HAVING COUNT(*) > 1;
```

2. **后端API校验中间件**
```java
// Spring Boot示例：响应体字段翻译键检查
@Aspect
@Component
public class TranslationCheckAspect {
  
  @Around("@annotation(org.springframework.web.bind.annotation.GetMapping)")
  public Object checkKeys(ProceedingJoinPoint joinPoint) throws Throwable {
    Object result = joinPoint.proceed();
    
    new ObjectMapper().acceptJsonFormatVisitor(result.getClass(), new JsonFormatVisitorWrapper() {
      @Override
      public void expectStringFormat(JavaType type) {
        String value = (String) result;
        if (!translationService.existsKey(value)) {
          throw new MissingTranslationException(value);
        }
      }
    });
    
    return result;
  }
}
```

#### **六、异常处理规范**
```typescript
// 统一不一致处理策略
class ConsistencyErrorHandler {
  static handle(errorType: 'NAMING'|'RELATION'|'TRANSLATION') {
    switch(errorType) {
      case 'NAMING':
        Sentry.captureException(new NamingError());
        return Response.error(409, '命名冲突');
        
      case 'RELATION':
        Database.rollback();
        Alert.sms('DBA紧急通知');
        return Response.error(500);
        
      case 'TRANSLATION':
        Cache.fallbackToDefaultLang();
        return Response.retryAfter(300);
    }
  }
}
```
