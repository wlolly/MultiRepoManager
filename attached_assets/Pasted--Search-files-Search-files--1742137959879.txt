- 建议：

async executeInventoryTransaction(transaction: InventoryTransaction) {
  return await db.transaction(async (trx) => {
    // 加锁查询当前库存
    const product = await trx.select()
      .from(products)
      .where(eq(products.id, transaction.productId))
      .forUpdate()
      .first();
    
    // 库存校验
    if (transaction.operationType === InventoryOperationType.Outbound) {
      if (product.stock + transaction.quantity < 0) {
        throw new Error('库存不足');
      }
    }
    
    // 更新库存
    await trx.update(products)
      .set({ stock: product.stock + transaction.quantity })
      .where(eq(products.id, transaction.productId));
      
    // 处理唯一码
    if (transaction.uniqueCode) {
      await this.processUniqueCodeTransaction(transaction, trx);
    }
    
    // 记录库存历史
    await trx.insert(inventoryHistory).values({
      productId: transaction.productId,
      warehouseId: transaction.warehouseId,
      beforeQuantity: product.stock,
      afterQuantity: product.stock + transaction.quantity,
      changeQuantity: transaction.quantity,
      operationType: transaction.operationType,
      operatorId: transaction.userId,
      operationTime: new Date(),
      reason: transaction.notes
    });
  });
}

```typescript
async executeTransfer(transferId: number, userId: number) {
  return await db.transaction(async (trx) => {
    // 锁定调拨单
    const transfer = await trx.select()
      .from(warehouseTransfers)
      .where(eq(warehouseTransfers.id, transferId))
      .forUpdate()
      .first();
    
    // 检查调拨单状态
    if (transfer.status !== 'pending') {
      throw new Error('调拨单状态不正确');
    }
    
    // 执行调拨出库
    for (const item of transfer.items) {
      await this.executeInventoryTransaction({
        warehouseId: transfer.sourceWarehouseId,
        productId: item.productId,
        quantity: -item.quantity,
        operationType: InventoryOperationType.TransferOut,
        referenceId: transferId,
        userId
      }, trx);
    }
    
    // 执行调拨入库
    for (const item of transfer.items) {
      await this.executeInventoryTransaction({
        warehouseId: transfer.targetWarehouseId,
        productId: item.productId,
        quantity: item.quantity,
        operationType: InventoryOperationType.TransferIn,
        referenceId: transferId,
        userId
      }, trx);
    }
    
    // 更新调拨单状态
    await trx.update(warehouseTransfers)
      .set({ status: 'completed' })
      .where(eq(warehouseTransfers.id, transferId));
  });
}
```

async checkInventoryAlert() {
  const alerts = await db.select()
    .from(productInventoryAlerts)
    .innerJoin(products, eq(products.id, productInventoryAlerts.productId))
    .where(
      or(
        lt(products.stock, productInventoryAlerts.minStock),
        gt(products.stock, productInventoryAlerts.maxStock)
      )
    );
    
  for (const alert of alerts) {
    // 发送预警通知
    await this.notificationService.sendInventoryAlert({
      productId: alert.productId,
      currentStock: alert.stock,
      threshold: alert.alertThreshold,
      warehouseId: alert.warehouseId,
      notifyUsers: alert.notifyUsers
    });
  }
}
